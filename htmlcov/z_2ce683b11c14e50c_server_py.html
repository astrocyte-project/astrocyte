1"""Shared FastMCP wiring for Astrocyte MCP servers (ADR-007, ADR-014). 

2 

3``AstrocyteMCP`` composes a FastMCP server with the actuation policy engine 

4and audit log so no server can expose a write tool that bypasses policy: write 

5tools call :meth:`enforce` with the action they are about to perform and only 

6proceed on an ``allowed`` decision. This class is the seed of the plugin SDK's 

7MCP base (#34). 

8""" 

9 

10from __future__ import annotations 

11 

12from typing import Any 

13 

14from fastmcp import FastMCP 

15 

16from astrocyte.core.policy import Action, Decision, DecisionStatus, PolicyEngine 

17 

18 

19class AstrocyteMCP: 

20 """A FastMCP server bound to Astrocyte's policy engine.""" 

21 

22 def __init__( 

23 self, 

24 name: str, 

25 policy: PolicyEngine, 

26 *, 

27 instructions: str | None = None, 

28 ) -> None: 

29 self.server = FastMCP(name=name, instructions=instructions) 

30 self.policy = policy 

31 # Re-exported decorator so subsystems register tools the FastMCP way. 

32 self.tool = self.server.tool 

33 

34 def enforce(self, action: Action, approval_token: str | None = None) -> Decision: 

35 """Authorize (and audit) one write action.""" 

36 return self.policy.authorize(action, approval_token) 

37 

38 @staticmethod 

39 def decision_payload(decision: Decision) -> dict[str, Any]: 

40 """The wire shape write tools return when a call is not executed.""" 

41 payload: dict[str, Any] = { 

42 "status": decision.status.value, 

43 "tier": decision.tier.value, 

44 "reason": decision.reason, 

45 } 

46 if ( 

47 decision.status is DecisionStatus.PENDING_APPROVAL 

48 and decision.approval is not None 

49 ): 

50 payload["approval_id"] = decision.approval.approval_id 

51 payload["expires_at"] = decision.approval.expires_at 

52 return payload 

53 

54 def http_app(self, path: str = "/") -> Any: 

55 """ASGI app for mounting into FastAPI (ADR-011 mounts at ``/mcp/ha``).""" 

56 return self.server.http_app(path=path)
